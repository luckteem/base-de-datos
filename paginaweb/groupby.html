3. GROUP BY

Para resumir datos.

Funciones: COUNT(), SUM(), AVG(), MAX(), MIN()

Contar total

SELECT COUNT(*) FROM clientes;

Contar por país

SELECT pais, COUNT(*) FROM clientes GROUP BY pais;

WHERE filtra filas, HAVING filtra grupos

SELECT pais, COUNT(*) as total FROM clientes GROUP BY pais HAVING total > 5;

Top 5

SELECT pais, COUNT(*) as total FROM clientes GROUP BY pais ORDER BY total DESC LIMIT 5;
